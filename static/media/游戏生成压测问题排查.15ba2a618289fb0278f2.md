## 查看哪些进程开销较大
```bash
ps -eww -o pid,ppid,comm,%mem,rss,vsz,args --sort=-%mem \
| awk '
NR == 1 {
    printf "%-8s %-8s %-20s %-6s %-10s %-10s %s\n",
           "PID", "PPID", "COMMAND", "%MEM", "RSS(MB)", "VSZ(MB)", "ARGS"
    next
}
{
    args = $7
    for (i = 8; i <= NF; i++) args = args " " $i

    if (length(args) > 120)
        args = substr(args, 1, 117) "..."

    printf "%-8s %-8s %-20s %-6s %-10.1f %-10.1f %s\n",
           $1, $2, $3, $4, $5 / 1024, $6 / 1024, args
}'
```
